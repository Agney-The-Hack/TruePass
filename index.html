📋 First-Time Supabase Setup (Run Once)

In your Supabase Dashboard, click SQL Editor → New Query, paste the SQL below, and click Run: 

-- 1. Create table for Enrolled People
CREATE TABLE IF NOT EXISTS public.faces (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    name TEXT UNIQUE NOT NULL,
    role TEXT DEFAULT '',
    notes TEXT DEFAULT '',
    photo_url TEXT DEFAULT '',
    descriptors JSONB NOT NULL,
    created_at TIMESTAMPTZ DEFAULT timezone('utc'::text, now()) NOT NULL,
    updated_at TIMESTAMPTZ DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 2. Create table for Recognition Logs
CREATE TABLE IF NOT EXISTS public.face_logs (
    id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name TEXT NOT NULL,
    confidence INTEGER,
    distance NUMERIC(6,4),
    photo_url TEXT DEFAULT '',
    created_at TIMESTAMPTZ DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 3. GRANT TABLE PERMISSIONS (Fixes permission denied error)
GRANT USAGE ON SCHEMA public TO anon, authenticated, service_role;
GRANT ALL ON TABLE public.faces TO anon, authenticated, service_role;
GRANT ALL ON TABLE public.face_logs TO anon, authenticated, service_role;
GRANT ALL ON ALL SEQUENCES IN SCHEMA public TO anon, authenticated, service_role;

-- 4. Enable Row Level Security (RLS) & Policies
ALTER TABLE public.faces ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.face_logs ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Allow public on faces" ON public.faces;
CREATE POLICY "Allow public on faces" ON public.faces FOR ALL TO anon, authenticated, service_role USING (true) WITH CHECK (true);
DROP POLICY IF EXISTS "Allow public on logs" ON public.face_logs;
CREATE POLICY "Allow public on logs" ON public.face_logs FOR ALL TO anon, authenticated, service_role USING (true) WITH CHECK (true);

-- 5. Storage Bucket for Photos
INSERT INTO storage.buckets (id, name, public)
VALUES ('face-photos', 'face-photos', true)
ON CONFLICT (id) DO UPDATE SET public = true;
DROP POLICY IF EXISTS "Allow public on photos" ON storage.objects;
CREATE POLICY "Allow public on photos" ON storage.objects FOR ALL TO anon, authenticated, service_role USING (bucket_id = 'face-photos') WITH CHECK (bucket_id = 'face-photos');

⚡ How to Build This Into Any Website

This facial recognition system is designed as a modular JavaScript SDK (`facial-recognition-sdk.js`). You can drop it into any HTML, React, Vue, or Angular website in under 5 minutes.

Step 1: Include the SDK and Dependencies in your HTML

Add the Face-API CDN and the SDK script inside your <head> or before </body>:

<!-- Face-API Neural Network Library -->
<script src="https://cdn.jsdelivr.net/npm/@vladmandic/face-api@1.7.12/dist/face-api.js"></script>

<!-- BioVision Facial Recognition SDK -->
<script src="js/face-recognition-sdk.js"></script>

Step 2: Initialize & Recognize in 5 Lines of JavaScript

// 1. Create SDK instance
const faceSDK = new FacialRecognitionSDK();
await faceSDK.init();

// 2. Start webcam on any <video> element
const video = document.getElementById('myWebcamVideo');
await faceSDK.startCamera(video);

// 3. Enroll a user with their image or current video frame
await faceSDK.enrollPerson('Alice', video, { role: 'Admin' });

// 4. Recognize whoever is in front of the camera
const results = await faceSDK.recognize(video);
results.forEach(person => {
    console.log(`Recognized: ${person.name} (${person.confidence}%)`);
});
