// Storage Persistence
function saveStateToStorage() {
  localStorage.setItem("truepass_user", JSON.stringify(state.user));
  localStorage.setItem("truepass_tickets", JSON.stringify(state.tickets));
}

function loadStateFromStorage() {
  const savedUser = localStorage.getItem("truepass_user");
  const savedTickets = localStorage.getItem("truepass_tickets");

  if (savedUser) {
    try {
      state.user = { ...state.user, ...JSON.parse(savedUser) };
    } catch (e) {}
  }
  if (savedTickets) {
    try {
      state.tickets = JSON.parse(savedTickets);
    } catch (e) {}
  }
}

// Navigation Tabs
function navigateTab(tabName) {
  // Hide all sections
  document.querySelectorAll(".tab-section").forEach((sec) => sec.classList.add("hidden"));

  // Reset tab button styles
  document.querySelectorAll("nav button").forEach((btn) => {
    btn.className =
      "px-4 py-1.5 rounded-lg text-slate-400 hover:text-white transition flex items-center space-x-2";
  });

  // Activate selected section & button
  const targetSection = document.getElementById(`section-${tabName}`);
  const targetNav = document.getElementById(`nav-${tabName}`);

  if (targetSection) targetSection.classList.remove("hidden");
  if (targetNav) {
    targetNav.className =
      "px-4 py-1.5 rounded-lg text-white bg-brand-500/20 text-brand-400 border border-brand-500/30 transition flex items-center space-x-2";
  }

  // Handle camera triggers on tab switches
  if (tabName === "biometrics" && !state.user.isEnrolled && !state.cameraStream) {
    startCamera();
  } else if (tabName !== "biometrics" && state.cameraStream) {
    stopCamera();
  }

  if (tabName === "gate") {
    setupGateTicketSelector();
    startGateCamera();
  } else if (state.gateCameraStream) {
    stopGateCamera();
  }

  if (tabName === "tickets") {
    renderTickets();
  }

  lucide.createIcons();
}

// BIOMETRICS CAMERA & RECOGNITION STUDIO
// -------------------------------------------------------------
async function startCamera() {
  const video = document.getElementById("webcam-video");
  const hudStatus = document.getElementById("hud-status-text");

  try {
    if (state.cameraStream) {
      stopCamera();
    }

    const stream = await navigator.mediaDevices.getUserMedia({
      video: {
        facingMode: "user",
        width: { ideal: 640 },
        height: { ideal: 480 },
      },
    });

    state.cameraStream = stream;
    video.srcObject = stream;
    video.play();

    document.getElementById("snapshot-preview-container").classList.add("hidden");
    document.getElementById("btn-capture-face").disabled = false;
    hudStatus.textContent = "Face Centered • Ready to Scan";
    hudStatus.className =
      "text-xs font-mono uppercase font-bold text-cyber-cyan px-2 py-1 rounded bg-black/60 border border-cyber-cyan/30 tracking-wider";

    startFaceHUDAnimation();
  } catch (err) {
    console.warn("Webcam access denied or unavailable:", err);
    hudStatus.textContent = "Camera Unavailable (Use Demo Profile)";
    hudStatus.className =
      "text-xs font-mono uppercase font-bold text-amber-400 px-2 py-1 rounded bg-black/60 border border-amber-500/30 tracking-wider";
  }
}

function stopCamera() {
  if (state.cameraStream) {
    state.cameraStream.getTracks().forEach((track) => track.stop());
    state.cameraStream = null;
  }
}

// Canvas HUD Animation Loop
let hudAnimationId = null;
function startFaceHUDAnimation() {
  const canvas = document.getElementById("face-canvas");
  if (!canvas) return;
  const ctx = canvas.getContext("2d");

  canvas.width = canvas.parentElement.clientWidth;
  canvas.height = canvas.parentElement.clientHeight;

  let angle = 0;
  function animate() {
    if (!state.cameraStream) return;

    ctx.clearRect(0, 0, canvas.width, canvas.height);
    const centerX = canvas.width / 2;
    const centerY = canvas.height / 2;

    // Outer rotating dashed rings
    ctx.save();
    ctx.translate(centerX, centerY);
    ctx.rotate(angle);
    ctx.strokeStyle = "rgba(0, 240, 255, 0.4)";
    ctx.lineWidth = 1.5;
    ctx.setLineDash([8, 8]);
    ctx.beginPath();
    ctx.arc(0, 0, Math.min(centerX, centerY) * 0.75, 0, Math.PI * 2);
    ctx.stroke();
    ctx.restore();

    // 4 Corner targeting markers
    angle += 0.015;
    hudAnimationId = requestAnimationFrame(animate);
  }

  cancelAnimationFrame(hudAnimationId);
  animate();
}

// Helper: Calculate SHA-256 via browser Web Crypto API
async function sha256Hex(message) {
  const msgBuffer = new TextEncoder().encode(message);
  const hashBuffer = await crypto.subtle.digest("SHA-256", msgBuffer);
  const hashArray = Array.from(new Uint8Array(hashBuffer));
  const hashHex = hashArray.map((b) => b.toString(16).padStart(2, "0")).join("");
  return `0x${hashHex}`;
}

// Capture Snapshot & Generate Biometric Hash
async function captureBiometricSnapshot() {
  const video = document.getElementById("webcam-video");
  const previewContainer = document.getElementById("snapshot-preview-container");
  const previewImg = document.getElementById("captured-snapshot-img");
  const previewHash = document.getElementById("preview-hash-display");

  // Create an offscreen canvas to capture the current video frame
  const snapCanvas = document.createElement("canvas");
  snapCanvas.width = 400;
  snapCanvas.height = 400;
  const ctx = snapCanvas.getContext("2d");

  // Center crop square
  const minDim = Math.min(video.videoWidth || 400, video.videoHeight || 400);
  const startX = ((video.videoWidth || 400) - minDim) / 2;
  const startY = ((video.videoHeight || 400) - minDim) / 2;

  // Mirror effect when drawing
  ctx.translate(400, 0);
  ctx.scale(-1, 1);
  ctx.drawImage(video, startX, startY, minDim, minDim, 0, 0, 400, 400);

  const snapshotDataUrl = snapCanvas.toDataURL("image/jpeg", 0.85);

  // Generate Salt & Commitment
  const attendeeSalt = `salt_${Date.now()}_${Math.random().toString(36).substring(2, 9)}`;
  const mockBiometricVector = `VEC_${snapshotDataUrl.slice(-100)}_${state.user.wallet}`;
  const biometricCommitment = await sha256Hex(`${mockBiometricVector}:${attendeeSalt}:${CLIENT_PEPPER}`);

  // Update State
  state.user.isEnrolled = true;
  state.user.faceSnapshot = snapshotDataUrl;
  state.user.salt = attendeeSalt;
  state.user.biometricHash = biometricCommitment;

  saveStateToStorage();
  updateUserUI();

  // Show captured preview
  previewImg.src = snapshotDataUrl;
  previewHash.textContent = biometricCommitment;
  previewContainer.classList.remove("hidden");

  stopCamera();

  // Show toast notification
  alert(`✅ Face Recognition Enrolled Successfully!\n\nYour Salted Biometric Commitment:\n${biometricCommitment}\n\nYou can now buy tickets.`);
}

// Simulate Razorpay Payment & Smart Contract Mint Relayer
async function executeRazorpayPayment() {
  const event = state.selectedEventForCheckout;
  if (!event) return;

  const confirmBtn = document.getElementById("btn-confirm-payment");
  const processingBox = document.getElementById("payment-processing");
  const statusText = document.getElementById("processing-status-text");

  confirmBtn.classList.add("hidden");
  processingBox.classList.remove("hidden");

  statusText.textContent = "Processing Razorpay Payment...";

  // Simulate network payment capture latency
  await new Promise((resolve) => setTimeout(resolve, 1400));

  statusText.textContent = "Payment Captured (INR)! Relayer minting ERC-721 token on-chain...";
  await new Promise((resolve) => setTimeout(resolve, 1600));

  // Generate on-chain ticket entity
  const nextTokenId = state.tickets.length + 1;
  const mockTxHash = `0x${Array.from({ length: 64 }, () => Math.floor(Math.random() * 16).toString(16)).join("")}`;

  const newTicket = {
    tokenId: nextTokenId,
    eventId: event.id,
    eventTitle: event.title,
    eventDate: event.date,
    venue: event.venue,
    tier: event.tier,
    price: event.price,
    owner: state.user.wallet,
    attendeeName: state.user.name,
    faceSnapshot: state.user.faceSnapshot,
    biometricHash: state.user.biometricHash,
    status: "ACTIVE", // ACTIVE | USED | CANCELLED
    mintedAt: Math.floor(Date.now() / 1000),
    txHash: mockTxHash,
    blockNumber: 19482030 + nextTokenId,
  };

  state.tickets.unshift(newTicket);
  saveStateToStorage();

  closeCheckoutModal();
  updateUserUI();
  navigateTab("tickets");

  alert(
    `🎉 Ticket Minted Successfully!\n\n` +
      `Token ID: #TPASS-${nextTokenId}\n` +
      `Bound Biometric Hash: ${newTicket.biometricHash.slice(0, 14)}...\n` +
      `On-Chain Tx: ${mockTxHash.slice(0, 16)}...\n\n` +
      `Your face is now cryptographically encoded into the ticket QR.`
  );
}

              <div class="mt-1 flex items-center space-x-1 text-[10px] font-mono text-brand-300/90 bg-brand-950/40 px-2 py-0.5 rounded border border-brand-500/20 truncate">
                <i data-lucide="fingerprint" class="w-3 h-3 text-brand-400 flex-shrink-0"></i>
                <span class="truncate" title="${t.biometricHash}">${t.biometricHash}</span>
              </div>
            </div>

          </div>
        </div>

        <!-- Bottom Section: Cryptographic Biometric QR Code -->
        <div class="relative z-10 pt-8 flex flex-col sm:flex-row items-center justify-between gap-4">
          <div class="text-left">
            <span class="text-[10px] font-mono text-slate-400 uppercase tracking-widest block">One-Time Scannable QR</span>
            <span class="text-xs text-slate-300 block mt-0.5">Contains on-chain Token ID & Salted Biometric Hash for physical gatekeeper match.</span>
            <span class="text-[10px] font-mono text-slate-500 mt-2 block">Tx: ${t.txHash.slice(0, 18)}...</span>
          </div>

          <!-- Dynamic QR Container -->
          <div class="bg-white p-2.5 rounded-2xl shadow-xl flex-shrink-0" id="ticket-qr-${t.tokenId}"></div>
        </div>

        <!-- Action Bar -->
        <div class="relative z-10 mt-6 pt-4 border-t border-slate-800 flex items-center justify-end space-x-3">
          ${
            !isUsed
              ? `<button onclick="sendTicketToGateScanner(${t.tokenId})" class="px-4 py-2 rounded-xl bg-purple-500/20 hover:bg-purple-500/30 text-purple-300 border border-purple-500/40 text-xs font-mono font-bold transition flex items-center space-x-1.5">
                   <i data-lucide="shield-check" class="w-3.5 h-3.5"></i>
                   <span>Test Gate Entry</span>
                 </button>`
              : `<span class="text-xs font-mono text-slate-500">Ticket Redeemed at Venue</span>`
          }
        </div>

      </div>
    `;
    })
    .join("");

  // Generate QR codes for each ticket
  state.tickets.forEach((t) => {
    const qrContainer = document.getElementById(`ticket-qr-${t.tokenId}`);
    if (qrContainer && typeof QRCode !== "undefined") {
      qrContainer.innerHTML = "";
      // Payload encoded into the QR code includes the biometric commitment!
      const qrPayload = JSON.stringify({
        v: "1.0",
        tokenId: t.tokenId,
        eventId: t.eventId,
        owner: t.owner,
        biometricHash: t.biometricHash,
        mintedAt: t.mintedAt,
        status: t.status,
      });

      new QRCode(qrContainer, {
        text: qrPayload,
        width: 110,
        height: 110,
        colorDark: "#0a0d14",
        colorLight: "#ffffff",
        correctLevel: QRCode.CorrectLevel.M,
      });
    }
  });

  lucide.createIcons();
}

function sendTicketToGateScanner(tokenId) {
  navigateTab("gate");
  selectTicketForGate(tokenId);
}

function setupGateTicketSelector() {
  const container = document.getElementById("gate-ticket-selector");
  if (!container) return;

  if (state.tickets.length === 0) {
    container.innerHTML = `<div class="text-xs text-slate-500 font-mono">No tickets purchased yet. Buy a ticket first to test gate redemption.</div>`;
    return;
  }

  container.innerHTML = state.tickets
    .map((t) => {
      const isSelected = state.selectedTicketForGate && state.selectedTicketForGate.tokenId === t.tokenId;
      return `
      <div onclick="selectTicketForGate(${t.tokenId})" class="p-3 rounded-2xl border ${
        isSelected ? "border-purple-500 bg-purple-500/10" : "border-slate-800 bg-slate-900/60"
      } cursor-pointer hover:border-slate-700 transition flex items-center justify-between">
        <div class="flex items-center space-x-3">
          <div class="w-10 h-10 rounded-xl overflow-hidden border border-slate-700 bg-black flex-shrink-0">
            <img src="${t.faceSnapshot}" class="w-full h-full object-cover" />
          </div>
          <div>
            <div class="text-xs font-bold text-white">#TPASS-${t.tokenId}: ${t.eventTitle}</div>
            <div class="text-[10px] font-mono text-slate-400">${t.attendeeName} • ${t.status}</div>
          </div>
        </div>
        <span class="text-[10px] font-mono ${t.status === "ACTIVE" ? "text-emerald-400" : "text-slate-500"} font-bold">
          ${t.status}
        </span>
      </div>
    `;
    })
    .join("");

  if (!state.selectedTicketForGate && state.tickets.length > 0) {
    selectTicketForGate(state.tickets[0].tokenId);
  }
}

function selectTicketForGate(tokenId) {
  const ticket = state.tickets.find((t) => t.tokenId === tokenId);
  if (!ticket) return;

  state.selectedTicketForGate = ticket;

  document.getElementById("gate-preview-title").textContent = `#TPASS-${ticket.tokenId} - ${ticket.eventTitle}`;
  document.getElementById("gate-preview-hash").textContent = ticket.biometricHash;

  const statusEl = document.getElementById("gate-preview-status");
  statusEl.textContent = ticket.status;
  statusEl.className =
    ticket.status === "ACTIVE"
      ? "px-2 py-0.5 rounded text-[10px] bg-emerald-500/20 text-emerald-300 font-mono"
      : "px-2 py-0.5 rounded text-[10px] bg-slate-800 text-slate-400 font-mono";

  setupGateTicketSelector();
  document.getElementById("gate-result-box").classList.add("hidden");
}

async function startGateCamera() {
  const video = document.getElementById("gate-webcam");
  if (!video) return;

  try {
    if (state.gateCameraStream) {
      stopGateCamera();
    }
    const stream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: "user" } });
    state.gateCameraStream = stream;
    video.srcObject = stream;
    video.play();
  } catch (e) {
    console.warn("Gate camera not accessible", e);
  }
}

function stopGateCamera() {
  if (state.gateCameraStream) {
    state.gateCameraStream.getTracks().forEach((t) => t.stop());
    state.gateCameraStream = null;
  }
}

// Simulate Gate Biometric Scan & State Transition (Active -> Used)
function simulateGateVerification(isLegitimateMatch) {
  const ticket = state.selectedTicketForGate;
  const resultBox = document.getElementById("gate-result-box");
  if (!ticket) {
    alert("Please select a ticket first.");
    return;
  }

  resultBox.classList.remove("hidden");

  // Case 1: Ticket Already Used
  if (ticket.status === "USED") {
    resultBox.className =
      "mt-6 p-6 rounded-3xl border border-rose-500/40 bg-rose-500/10 text-center animate-scale-up";
    resultBox.innerHTML = `
      <div class="w-12 h-12 rounded-2xl bg-rose-500/20 text-rose-400 flex items-center justify-center mx-auto mb-3">
        <i data-lucide="shield-alert" class="w-6 h-6"></i>
      </div>
      <h3 class="text-lg font-black text-rose-400">ENTRY DENIED — TICKET ALREADY USED</h3>
      <p class="text-xs text-slate-300 mt-1 max-w-md mx-auto">
        Token #TPASS-${ticket.tokenId} has already been redeemed at the venue entrance. Re-entry or secondary scanning is blocked by contract state.
      </p>
    `;
    lucide.createIcons();
    return;
  }

  // Case 2: Fraudster / Imposter Face Mismatch
  if (!isLegitimateMatch) {
    resultBox.className =
      "mt-6 p-6 rounded-3xl border border-amber-500/40 bg-amber-500/10 text-center animate-scale-up";
    resultBox.innerHTML = `
      <div class="w-12 h-12 rounded-2xl bg-amber-500/20 text-amber-400 flex items-center justify-center mx-auto mb-3">
        <i data-lucide="user-x" class="w-6 h-6"></i>
      </div>
      <h3 class="text-lg font-black text-amber-400">ENTRY REJECTED — BIOMETRIC MISMATCH</h3>
      <p class="text-xs text-slate-300 mt-1 max-w-md mx-auto">
        The face scanned at the turnstile does not match the salted biometric commitment (<code class="text-amber-300 font-mono">${ticket.biometricHash.slice(0, 14)}...</code>) bound to this ticket on-chain. Fraud prevention triggered.
      </p>
    `;
    lucide.createIcons();
    return;
  }

  // Case 3: Legitimate Match (Success -> Active to Used)
  ticket.status = "USED";
  ticket.redeemedAt = Math.floor(Date.now() / 1000);
  saveStateToStorage();

  selectTicketForGate(ticket.tokenId);

  resultBox.className =
    "mt-6 p-6 rounded-3xl border border-emerald-500/40 bg-emerald-500/10 text-center animate-scale-up";
  resultBox.innerHTML = `
    <div class="w-12 h-12 rounded-2xl bg-emerald-500/20 text-emerald-400 flex items-center justify-center mx-auto mb-3">
      <i data-lucide="check-circle" class="w-6 h-6"></i>
    </div>
    <h3 class="text-lg font-black text-emerald-400">ACCESS GRANTED — WELCOME TO THE EVENT</h3>
    <p class="text-xs text-slate-300 mt-1 max-w-md mx-auto">
      Face recognized & validated against on-chain hash! Smart contract executed <code class="text-emerald-300 font-mono">redeemTicket(${ticket.tokenId})</code>. State transitioned: <strong class="text-white">Active → Used</strong>.
    </p>
  `;
  lucide.createIcons();
}
